REM *** Read ..\docs\TIMING_NOTES.md before changing the synth_ecp5 command
REM *** below (the "scratchpad -set abc9.xaiger 1" bit works around a real
REM *** yosys crash bug, see notes #3) or the nextpnr-ecp5 seed sweep (this
REM *** design is timing-marginal by design at 1024x768, see notes #2 --
REM *** the seed sweep is not decorative, it's load-bearing).
setlocal
echo BUILD STARTED
pause
REM --- define project name ---
set PROJ=video

REM --- find repo root (this script lives in tools/) ---
cd /d "%~dp0.."

REM --- generate palette from OKLCH data ---
echo Generating palette from OKLCH...
python tools\fpga_conversion_scripts\oklch_to_rgb.py || exit /b 1
python tools\fpga_conversion_scripts\gen_color_bin.py || exit /b 1

REM --- move into src directory for synthesis ---
cd src

REM --- run synthesis (NO -q so errors show) ---
REM Recent yosys builds default synth_ecp5 to the ABC9 flow, whose
REM experimental write_xaiger2 writer can crash with an internal assert on
REM some designs (unrelated to your Verilog). The scratchpad setting below
REM is documented upstream as the "chicken bit" that forces ABC9 to use the
REM older, stable AIGER writer instead of the buggy experimental one.
yosys -q -p "scratchpad -set abc9.xaiger 1; read_verilog top.v tmds_encoder.v pll.v; synth_ecp5 -top top -json video.json" || exit /b 1

REM --- place & route ---
REM This design's 325.1 MHz shift-clock requirement is right on the edge of
REM what ECP5 -6 can close (recent runs landed at 320-324 MHz vs the 325.1
REM MHz target -- within ~0.2-1.5%). nextpnr's placer is randomized, so a
REM different --seed can be the difference between pass and fail on the
REM exact same source. Try a few seeds and keep the first one that closes;
REM only fall back to --timing-allow-fail if none of them do.
echo Running nextpnr-ecp5 (default seed)...
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config
if not errorlevel 1 goto pnr_ok

echo Default seed missed timing, trying seed 1...
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --seed 1 --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config
if not errorlevel 1 goto pnr_ok

echo Seed 1 missed timing, trying seed 2...
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --seed 2 --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config
if not errorlevel 1 goto pnr_ok

echo Seed 2 missed timing, trying seed 3...
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --seed 3 --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config
if not errorlevel 1 goto pnr_ok

echo Seed 3 missed timing, trying seed 4...
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --seed 4 --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config
if not errorlevel 1 goto pnr_ok

echo All seed attempts missed the 325.1 MHz target.
echo Falling back to --timing-allow-fail so you still get a bitstream to test with --
echo this is fine for visual bring-up, but don't treat it as production-ready
echo until a run actually closes timing without this flag.
nextpnr-ecp5 --85k --package CABGA381 --speed 6 --timing-allow-fail --json %PROJ%.json --lpf ulx3s.lpf --textcfg %PROJ%.config || exit /b 1

:pnr_ok

REM --- pack bitstream ---
ecppack %PROJ%.config %PROJ%.bit || exit /b 1

REM --- flash or load ---
if /i "%1"=="flash" (
    openFPGALoader -b ulx3s -f --unprotect-flash %PROJ%.bit
) else (
    openFPGALoader -b ulx3s %PROJ%.bit
)

echo BUILD COMPLETE
pause

// *** Read docs/TIMING_NOTES.md before changing anything in this file's
// *** clocking/timing/serializer section (h_cnt/v_cnt limits, the
// *** clk_shift-domain always block, or the ODDRX1F instances). The
// *** shift clock is fixed at exactly 5x the pixel clock by the serializer;
// *** changing resolution without also retuning pll.v will break the video
// *** output. At 800x600 the shift clock is 200 MHz, which has plenty of
// *** margin on ECP5 -6 (the old 1024x768 / 325 MHz setup did not).
// ***
// *** The palette include below (palette_pkg.vh) is a build-time constant
// *** on purpose -- see TIMING_NOTES.md #4 before changing it back to a
// *** $readmemh-loaded register.
module top (
    input  wire       clk_25mhz,
    output wire [3:0] gpdi_dp      // [0]=blue [1]=green [2]=red [3]=clock
);
    // ---------- clocks ----------
    wire clk_shift, clk_pixel, locked;
    pll pll_i (.clkin(clk_25mhz), .clkout0(clk_shift),
               .clkout1(clk_pixel), .locked(locked));

    // ---------- external palette (OKLCH->RGB result) ----------
    // Generated by tools/fpga_conversion_scripts/gen_color_bin.py from
    // data/oklch/theme.ini. This is a build-time constant (it only changes
    // when you rerun the python step and resynthesize), so it's pulled in
    // as a `localparam via `include rather than loaded into a register with
    // $readmemh at run time. That lets yosys constant-fold it away instead
    // of spending real LUTs/DFFs/muxes on it.
    `include "../build/palette_pkg.vh"

    wire [7:0] white_r = PALETTE_WHITE[23:16];
    wire [7:0] white_g = PALETTE_WHITE[15:8];
    wire [7:0] white_b = PALETTE_WHITE[7:0];

    // ---------- 800x600 @ 60 Hz timing (VESA, 40 MHz pixel clock) ----------
    // Both syncs are POSITIVE polarity for this mode (the old 1024x768 mode
    // used negative polarity, so don't copy the ~() inversion back in).
    localparam H_ACTIVE = 800, H_FRONT = 40, H_SYNC = 128, H_BACK = 88;
    localparam V_ACTIVE = 600, V_FRONT = 1,  V_SYNC = 4,   V_BACK = 23;

    localparam H_TOTAL      = H_ACTIVE + H_FRONT + H_SYNC + H_BACK;  // 1056
    localparam V_TOTAL      = V_ACTIVE + V_FRONT + V_SYNC + V_BACK;  // 628
    localparam H_SYNC_START = H_ACTIVE + H_FRONT;                    // 840
    localparam H_SYNC_END   = H_SYNC_START + H_SYNC;                 // 968
    localparam V_SYNC_START = V_ACTIVE + V_FRONT;                    // 601
    localparam V_SYNC_END   = V_SYNC_START + V_SYNC;                 // 605

    reg [10:0] h_cnt = 0;
    reg [10:0] v_cnt = 0;

    always @(posedge clk_pixel) begin
        if (h_cnt == H_TOTAL - 1) begin
            h_cnt <= 0;
            if (v_cnt == V_TOTAL - 1) v_cnt <= 0;
            else                      v_cnt <= v_cnt + 1;
        end else begin
            h_cnt <= h_cnt + 1;
        end
    end

    wire h_active = (h_cnt < H_ACTIVE);
    wire v_active = (v_cnt < V_ACTIVE);
    wire hsync    =  (h_cnt >= H_SYNC_START && h_cnt < H_SYNC_END);
    wire vsync    =  (v_cnt >= V_SYNC_START && v_cnt < V_SYNC_END);
    wire blank    = ~(h_active && v_active);

    // ---------- "desktop" + "work surface" layout ----------
    // The whole 800x600 active area is the desktop, filled with the palette
    // color (white from build/palette_pkg.vh). A 640x480 work surface sits
    // centered on top of it in a fixed dark blue, so the desktop stays
    // visible as a 80 px left/right and 60 px top/bottom margin.
    localparam WORK_W  = 640;
    localparam WORK_H  = 480;
    localparam WORK_X0 = (H_ACTIVE - WORK_W) / 2;   // 80
    localparam WORK_Y0 = (V_ACTIVE - WORK_H) / 2;   // 60
    localparam [23:0] WORK_RGB = 24'h000060;        // dark blue (hard-coded for now)

    wire in_work = (h_cnt >= WORK_X0) && (h_cnt < WORK_X0 + WORK_W) &&
                   (v_cnt >= WORK_Y0) && (v_cnt < WORK_Y0 + WORK_H);

    wire [7:0] pix_r = in_work ? WORK_RGB[23:16] : white_r;
    wire [7:0] pix_g = in_work ? WORK_RGB[15:8]  : white_g;
    wire [7:0] pix_b = in_work ? WORK_RGB[7:0]   : white_b;

    // ---------- TMDS encoding ----------
    wire [9:0] tmds_b, tmds_g, tmds_r;
    tmds_encoder enc_b (.clk(clk_pixel), .data(pix_b), .c({vsync, hsync}), .blank(blank), .encoded(tmds_b));
    tmds_encoder enc_g (.clk(clk_pixel), .data(pix_g), .c(2'b00),          .blank(blank), .encoded(tmds_g));
    tmds_encoder enc_r (.clk(clk_pixel), .data(pix_r), .c(2'b00),          .blank(blank), .encoded(tmds_r));

    // ---------- pixel-clock toggle, used to lock the shift phase ----------
    reg ptog = 0;
    always @(posedge clk_pixel) ptog <= ~ptog;

    // ---------- 10:2 serializer at 5x pixel clock (200 MHz) ----------
    reg       t1 = 0, t2 = 0;
    reg [4:0] ring = 5'b00001;
    reg [9:0] sr_b = 0, sr_g = 0, sr_r = 0, sr_c = 0;
    wire detect = (t1 != t2);

    always @(posedge clk_shift) begin
        t1 <= ptog;
        t2 <= t1;

        // one-hot ring, re-locked to the pixel clock every pixel
        if (detect) ring <= 5'b00001;
        else        ring <= {ring[3:0], ring[4]};

        if (ring[0]) begin
            sr_b <= tmds_b;
            sr_g <= tmds_g;
            sr_r <= tmds_r;
            sr_c <= 10'b0000011111;
        end else begin
            sr_b <= {2'b00, sr_b[9:2]};
            sr_g <= {2'b00, sr_g[9:2]};
            sr_r <= {2'b00, sr_r[9:2]};
            sr_c <= {2'b00, sr_c[9:2]};
        end
    end

    ODDRX1F ddr_b (.SCLK(clk_shift), .RST(1'b0), .D0(sr_b[0]), .D1(sr_b[1]), .Q(gpdi_dp[0]));
    ODDRX1F ddr_g (.SCLK(clk_shift), .RST(1'b0), .D0(sr_g[0]), .D1(sr_g[1]), .Q(gpdi_dp[1]));
    ODDRX1F ddr_r (.SCLK(clk_shift), .RST(1'b0), .D0(sr_r[0]), .D1(sr_r[1]), .Q(gpdi_dp[2]));
    ODDRX1F ddr_c (.SCLK(clk_shift), .RST(1'b0), .D0(sr_c[0]), .D1(sr_c[1]), .Q(gpdi_dp[3]));
endmodule

// diamond 3.7 accepts this PLL
// diamond 3.8-3.9 is untested
// diamond 3.10 or higher is likely to abort with error about unable to use feedback signal
// cause of this could be from wrong CPHASE/FPHASE parameters
//
// *** DO NOT change CLKOP_DIV/CLKOS_DIV/CLKFB_DIV/CLKI_DIV without reading
// *** docs/TIMING_NOTES.md first. clkout0 (shift clock) MUST stay at
// *** exactly 5x clkout1 (pixel clock) -- that ratio is fixed by the
// *** ODDRX1F-based serializer in top.v, not a free parameter. Changing
// *** resolution means recomputing BOTH this PLL's dividers AND top.v's
// *** timing constants together -- see the notes for a table of safe
// *** resolution/frequency targets.
//
// Current target: 800x600 @ 60 Hz (VESA)
//   pixel clock  =  40 MHz
//   shift clock  = 200 MHz (5x pixel)
// Generated with: ecppll -i 25 --clkout0 200 --clkout1 40
//   VCO = 25 MHz / CLKI_DIV(1) * CLKFB_DIV(8) * CLKOP_DIV(3) = 600 MHz
//   (ECP5 VCO range is 400-800 MHz, so 600 is comfortably mid-range)
//   clkout0 = VCO / CLKOP_DIV(3)  = 200 MHz
//   clkout1 = VCO / CLKOS_DIV(15) =  40 MHz
//
// (Previous 1024x768 values, for reference: CLKFB_DIV 13, CLKOP_DIV 2,
//  CLKOS_DIV 10 -> 325 MHz / 65 MHz, VCO 650 MHz.)
module pll
(
    input clkin, // 25 MHz, 0 deg
    output clkout0, // 200 MHz, 0 deg
    output clkout1, // 40 MHz, 0 deg
    output locked
);
(* FREQUENCY_PIN_CLKI="25" *)
(* FREQUENCY_PIN_CLKOP="200" *)
(* FREQUENCY_PIN_CLKOS="40" *)
(* ICP_CURRENT="12" *) (* LPF_RESISTOR="8" *) (* MFG_ENABLE_FILTEROPAMP="1" *) (* MFG_GMCREF_SEL="2" *)
EHXPLLL #(
        .PLLRST_ENA("DISABLED"),
        .INTFB_WAKE("DISABLED"),
        .STDBY_ENABLE("DISABLED"),
        .DPHASE_SOURCE("DISABLED"),
        .OUTDIVIDER_MUXA("DIVA"),
        .OUTDIVIDER_MUXB("DIVB"),
        .OUTDIVIDER_MUXC("DIVC"),
        .OUTDIVIDER_MUXD("DIVD"),
        .CLKI_DIV(1),
        .CLKOP_ENABLE("ENABLED"),
        .CLKOP_DIV(3),
        .CLKOP_CPHASE(1),
        .CLKOP_FPHASE(0),
        .CLKOS_ENABLE("ENABLED"),
        .CLKOS_DIV(15),
        .CLKOS_CPHASE(1),
        .CLKOS_FPHASE(0),
        .FEEDBK_PATH("CLKOP"),
        .CLKFB_DIV(8)
    ) pll_i (
        .RST(1'b0),
        .STDBY(1'b0),
        .CLKI(clkin),
        .CLKOP(clkout0),
        .CLKOS(clkout1),
        .CLKFB(clkout0),
        .CLKINTFB(),
        .PHASESEL0(1'b0),
        .PHASESEL1(1'b0),
        .PHASEDIR(1'b1),
        .PHASESTEP(1'b1),
        .PHASELOADREG(1'b1),
        .PLLWAKESYNC(1'b0),
        .ENCLKOP(1'b0),
        .LOCK(locked)
	);
endmodule
